const timeline=gsap.timeline({paused:true}); /* Create a rolling sequence: */ timeline .to('.die', {rotationX:360, rotationY:360, duration:0.5, stagger:0.1, ease:'power2.inOut'}) .to('.die', {scale:1.2, duration:0.2, stagger:0.05, ease:'elastic.out(1.5,0.5)'},'-=0.2') .to('.die', {scale:1, duration:0.3, stagger:0.05, ease:'back.out(1.7)'},'-=0.1'); timeline.play();
